# LearnMate backend dependencies.

# was built and verified against -- loosen them if you must, but read the notes first.

# --- Web layer -------------------------------------------------------------------------

uvicorn[standard]>=0.30
# Multipart form parsing. FastAPI needs it for UploadFile and will not start without it.
python-multipart>=0.0.9
PyJWT>=2.8,<3
bcrypt>=4.1
# Pydantic's EmailStr, used on the register and login models.
email-validator>=2.1

# --- Agent framework ---------------------------------------------------------------------

langchain-community>=0.4,<1

# --- Local inference ---------------------------------------------------------------------
# llama-cpp-python must be built with grammar support (any 0.2.77+ release is). Every
# structured output in this project depends on JSON-schema-constrained decoding; without
# it a 3B model returns prose where JSON was asked for about half the time.
llama-cpp-python>=0.3,<0.4

# --- Cloud inference (optional) -----------------------------------------------------------
# Only imported when LEARNMATE_GENERATOR_BACKEND or LEARNMATE_JUDGE_BACKEND is "gemini".
# Uninstall it if you are running fully offline; nothing else references it.

# --- Embeddings ---------------------------------------------------------------------------

# --- Storage ------------------------------------------------------------------------------

# for the local mongodb://localhost:27018 default, but harmless and one less thing to
# discover the hard way if this is ever pointed at Atlas.
pymongo>=4.9,<5
dnspython>=2.7

# Keep this within one minor version of the qdrant/qdrant image tag in docker-compose.yml
# -- the client refuses to talk to a server further away than that.

# --- PDF handling --------------------------------------------------------------------------
