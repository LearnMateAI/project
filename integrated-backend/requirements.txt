# LearnMate backend dependencies.
#
# Two halves: the web server, and the AI engine. The engine's pins are the ones this code
# was built and verified against -- loosen them if you must, but read the notes first.

# --- Web layer -------------------------------------------------------------------------
fastapi>=0.115,<1
uvicorn[standard]>=0.30
# Multipart form parsing. FastAPI needs it for UploadFile and will not start without it.
python-multipart>=0.0.9
# The [crypto] extra pulls in `cryptography`, which PyJWT needs for RS256 -- the
# algorithm Keycloak signs with. Without it PyJWKClient imports fine and only fails at
# the first verification. It is an extra rather than a bare dependency because the
# original HS256 login path does not need it; Keycloak verification does.
PyJWT[crypto]>=2.8,<3
bcrypt>=4.1
# Pydantic's EmailStr, used on the register and login models.
email-validator>=2.1

# --- Agent framework ---------------------------------------------------------------------
langchain>=1.3,<2
langchain-core>=1.5,<2
langchain-community>=0.4,<1
langchain-text-splitters>=1.1,<2
langgraph>=1.2,<2

# --- Local inference ---------------------------------------------------------------------
# llama-cpp-python must be built with grammar support (any 0.2.77+ release is). Every
# structured output in this project depends on JSON-schema-constrained decoding; without
# it a 3B model returns prose where JSON was asked for about half the time.
llama-cpp-python>=0.3,<0.4

# --- Cloud inference (optional) -----------------------------------------------------------
# Only imported when LEARNMATE_GENERATOR_BACKEND or LEARNMATE_JUDGE_BACKEND is "gemini".
# Uninstall it if you are running fully offline; nothing else references it.
google-genai>=1.0

# --- Building the finetuned GGUF (optional, build-time only) ------------------------------
# Needed only to run scripts/build_finetuned_gguf.py, which merges the model-Thevindu LoRA
# adapter into its base and converts the result to GGUF. Deliberately left commented: the
# server never imports either package, the build is a one-off, and its output is a plain
# file in models/ that needs nothing installed to load. Install them when you build:
#
#     pip install "peft>=0.20" "gguf>=0.10" sentencepiece
#
# torch and transformers are already required above (sentence-transformers pulls them in),
# so those three are the whole additional cost.
#
# sentencepiece is needed even though Qwen2.5 uses BPE, not sentencepiece: the converter's
# Qwen class calls _set_vocab_sentencepiece() first and falls back to BPE when it finds no
# tokenizer.model. That fallback catches FileNotFoundError, so without the package the
# import raises ImportError and the conversion dies before it can fall back.
# peft>=0.20
# gguf>=0.10
# sentencepiece

# --- Embeddings ---------------------------------------------------------------------------
sentence-transformers>=5.0,<6
torch>=2.6
transformers>=5.0

# --- Storage ------------------------------------------------------------------------------
# dnspython is required for mongodb+srv:// URIs; pymongo does not pull it in. Not needed
# for the local mongodb://localhost:27018 default, but harmless and one less thing to
# discover the hard way if this is ever pointed at Atlas.
pymongo>=4.9,<5
dnspython>=2.7

# Keep this within one minor version of the qdrant/qdrant image tag in docker-compose.yml
# -- the client refuses to talk to a server further away than that.
qdrant-client>=1.15,<2

# --- PDF handling --------------------------------------------------------------------------
# One PDF library for the whole system: extraction, cleaning and upload validation all go
# through PyMuPDF.
pymupdf>=1.26

# --- Office export (resources already generated; never re-runs the model) -----------------
python-docx>=1.1
python-pptx>=1.0

# --- Support -------------------------------------------------------------------------------
numpy>=2.0
python-dotenv>=1.0
huggingface_hub>=0.20
# Used only by scripts/smoke_test.py.
requests>=2.31
